cmake_minimum_required(VERSION 3.16)
project(vigil VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIGIL_TSAN "Enable thread sanitizer (separate build)" OFF)
option(VIGIL_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(VIGIL_FUZZER "Build Clang libFuzzer decoder target with ASan/UBSan" OFF)
option(VIGIL_BENCHMARKS "Build standalone measurement probes (not installed)" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wconversion
                    -Wsign-conversion -Wshadow -Wstrict-prototypes
                    -Wmissing-prototypes -Wformat=2 -Wundef -Wcast-align
                    -Wwrite-strings -Wvla -Wdouble-promotion -Wnull-dereference)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  add_compile_options(-Wformat-signedness -Wduplicated-cond -Wduplicated-branches -Wlogical-op)
endif()
if(VIGIL_TSAN AND VIGIL_SANITIZERS)
  message(FATAL_ERROR "Use separate sanitizer builds")
endif()
if(VIGIL_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR VIGIL_TSAN)
    message(FATAL_ERROR "libFuzzer requires a separate Clang build without TSan")
  endif()
  add_executable(fuzz-decoder fuzz/decoder.c src/protocol.c)
  target_include_directories(fuzz-decoder PRIVATE include)
  # Compile the decoder itself with coverage and sanitizers, not only the driver.
  target_compile_options(fuzz-decoder PRIVATE -UNDEBUG -fsanitize=fuzzer,address,undefined
                         -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(fuzz-decoder PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
if(VIGIL_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(VIGIL_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(vigil_protocol src/protocol.c)
target_include_directories(vigil_protocol PUBLIC include)
add_library(vigil_monitor src/monitor.c src/config.c)
target_link_libraries(vigil_monitor PUBLIC vigil_protocol)
target_compile_definitions(vigil_monitor PRIVATE _POSIX_C_SOURCE=200809L)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(vigil-decode tools/decode.c)
  target_link_libraries(vigil-decode PRIVATE vigil_protocol)
  target_compile_definitions(vigil-decode PRIVATE _POSIX_C_SOURCE=200809L)
  list(PREPEND CMAKE_PREFIX_PATH "/opt/vigil-sqlite-3.51.3")
  find_package(SQLite3 3.51.3 REQUIRED)
  find_package(PostgreSQL REQUIRED)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
  add_library(vigil_queue src/queue.c)
  target_include_directories(vigil_queue PUBLIC include)
  target_compile_definitions(vigil_queue PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(vigil_queue PUBLIC Threads::Threads)
  add_library(vigil_outbox src/outbox.c)
  target_compile_definitions(vigil_outbox PRIVATE _GNU_SOURCE)
  target_link_libraries(vigil_outbox PUBLIC vigil_queue PRIVATE SQLite::SQLite3)
  add_executable(vigil-outbox-init tools/outbox_init.c)
  target_link_libraries(vigil-outbox-init PRIVATE vigil_outbox)
  add_library(vigil_persistence src/persistence.c src/replay.c)
  target_compile_definitions(vigil_persistence PRIVATE _GNU_SOURCE)
  target_link_libraries(vigil_persistence PUBLIC vigil_outbox PRIVATE PostgreSQL::PostgreSQL)
  add_executable(vigild src/vigild.c)
  target_link_libraries(vigild PRIVATE vigil_monitor vigil_persistence PkgConfig::SYSTEMD)
  target_compile_definitions(vigild PRIVATE _GNU_SOURCE)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CMOCKA REQUIRED IMPORTED_TARGET cmocka)
  add_executable(test_protocol tests/test_protocol.c)
  target_link_libraries(test_protocol PRIVATE vigil_protocol PkgConfig::CMOCKA)
  target_compile_definitions(test_protocol PRIVATE
    VIGIL_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_vectors.txt")
  add_test(NAME protocol COMMAND test_protocol)
  add_executable(test_monitor tests/test_monitor.c)
  target_link_libraries(test_monitor PRIVATE vigil_monitor PkgConfig::CMOCKA)
  target_compile_definitions(test_monitor PRIVATE _POSIX_C_SOURCE=200809L)
  add_test(NAME monitor COMMAND test_monitor)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(vigil_outbox_test src/outbox.c tests/failpoint.c tests/sqlite_fault_vfs.c tests/sqlite_usage_faults.c)
    target_compile_definitions(vigil_outbox_test PRIVATE _GNU_SOURCE VIGIL_TEST_FAILPOINTS)
    target_link_libraries(vigil_outbox_test PUBLIC vigil_queue PRIVATE SQLite::SQLite3)
    target_link_options(vigil_outbox_test INTERFACE -Wl,--wrap=sqlite3_prepare_v2 -Wl,--wrap=sqlite3_step)
    add_library(vigil_persistence_test src/persistence.c src/replay.c)
    target_compile_definitions(vigil_persistence_test PRIVATE _GNU_SOURCE VIGIL_TEST_FAILPOINTS)
    target_link_libraries(vigil_persistence_test PUBLIC vigil_outbox_test PRIVATE PostgreSQL::PostgreSQL)
    add_executable(vigild-resilience-test src/vigild.c)
    target_compile_definitions(vigild-resilience-test PRIVATE _GNU_SOURCE)
    target_link_libraries(vigild-resilience-test PRIVATE vigil_monitor vigil_persistence_test PkgConfig::SYSTEMD)
    add_executable(test_outbox tests/test_outbox.c)
    target_compile_definitions(test_outbox PRIVATE _GNU_SOURCE VIGIL_TEST_FAILPOINTS)
    target_link_libraries(test_outbox PRIVATE vigil_outbox_test vigil_persistence_test SQLite::SQLite3 PkgConfig::CMOCKA)
    add_test(NAME outbox COMMAND test_outbox)
    add_executable(test_queue tests/test_queue.c)
    target_link_libraries(test_queue PRIVATE vigil_queue PkgConfig::CMOCKA)
    target_compile_definitions(test_queue PRIVATE _POSIX_C_SOURCE=200809L)
    add_test(NAME queue COMMAND test_queue)
  endif()
endif()
if(VIGIL_BENCHMARKS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(benchmark-core tests/benchmark_core.c)
  target_compile_definitions(benchmark-core PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(benchmark-core PRIVATE vigil_monitor)
  add_executable(benchmark-outbox tests/benchmark_outbox.c)
  target_compile_definitions(benchmark-outbox PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(benchmark-outbox PRIVATE vigil_outbox SQLite::SQLite3)
  target_link_options(benchmark-outbox PRIVATE -Wl,--wrap=sqlite3_exec)
endif()
