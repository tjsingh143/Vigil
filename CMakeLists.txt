cmake_minimum_required(VERSION 3.16)
project(vigil VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIGIL_TSAN "Enable thread sanitizer (separate build)" OFF)
option(VIGIL_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wconversion
                    -Wsign-conversion -Wshadow -Wstrict-prototypes
                    -Wmissing-prototypes -Wformat=2 -Wundef)
if(VIGIL_TSAN AND VIGIL_SANITIZERS)
  message(FATAL_ERROR "Use separate sanitizer builds")
endif()
if(VIGIL_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(VIGIL_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(vigil_protocol src/protocol.c)
target_include_directories(vigil_protocol PUBLIC include)
add_library(vigil_monitor src/monitor.c src/config.c)
target_link_libraries(vigil_monitor PUBLIC vigil_protocol)
target_compile_definitions(vigil_monitor PRIVATE _POSIX_C_SOURCE=200809L)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(vigil-decode tools/decode.c)
  target_link_libraries(vigil-decode PRIVATE vigil_protocol)
  target_compile_definitions(vigil-decode PRIVATE _POSIX_C_SOURCE=200809L)
  find_package(PostgreSQL REQUIRED)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
  add_library(vigil_queue src/queue.c)
  target_include_directories(vigil_queue PUBLIC include)
  target_compile_definitions(vigil_queue PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(vigil_queue PUBLIC Threads::Threads)
  add_library(vigil_persistence src/persistence.c)
  target_compile_definitions(vigil_persistence PRIVATE _GNU_SOURCE)
  target_link_libraries(vigil_persistence PUBLIC vigil_queue PRIVATE PostgreSQL::PostgreSQL)
  add_executable(vigild src/vigild.c)
  target_link_libraries(vigild PRIVATE vigil_monitor vigil_persistence PkgConfig::SYSTEMD)
  target_compile_definitions(vigild PRIVATE _GNU_SOURCE)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CMOCKA REQUIRED IMPORTED_TARGET cmocka)
  add_executable(test_protocol tests/test_protocol.c)
  target_link_libraries(test_protocol PRIVATE vigil_protocol PkgConfig::CMOCKA)
  target_compile_definitions(test_protocol PRIVATE
    VIGIL_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_vectors.txt")
  add_test(NAME protocol COMMAND test_protocol)
  add_executable(test_monitor tests/test_monitor.c)
  target_link_libraries(test_monitor PRIVATE vigil_monitor PkgConfig::CMOCKA)
  target_compile_definitions(test_monitor PRIVATE _POSIX_C_SOURCE=200809L)
  add_test(NAME monitor COMMAND test_monitor)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_queue tests/test_queue.c)
    target_link_libraries(test_queue PRIVATE vigil_queue PkgConfig::CMOCKA)
    target_compile_definitions(test_queue PRIVATE _POSIX_C_SOURCE=200809L)
    add_test(NAME queue COMMAND test_queue)
  endif()
endif()
